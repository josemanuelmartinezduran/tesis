# Arquitectura de Red Neuronal Paramétrica para Tesis de Melodías
# Cada fila representa una capa con sus parámetros separados por comas:
# [tipo_capa, param1, param2, param3, param4]
#
# Tipos de capas:
#   0: LAYER_NONE (Ignorada/omitida, útil para cromosomas de tamaño fijo pero profundidad variable)
#   1: LAYER_DENSE (Dense) -> [1, units, activation_code]
#   2: LAYER_CONV2D (Conv2D) -> [2, filters, kernel_size, activation_code, padding_code]
#   3: LAYER_LSTM (LSTM) -> [3, units, activation_code, return_sequences]
#   4: LAYER_DROPOUT (Dropout) -> [4, rate]
#   5: LAYER_FLATTEN (Flatten) -> [5]
#   6: LAYER_CONV1D (Conv1D) -> [6, filters, kernel_size, activation_code, padding_code]
#
# Mapeo de Códigos de Activación:
#   0: linear, 1: relu, 2: tanh, 3: sigmoid, 4: softmax, 5: selu, 6: elu
# Mapeo de Códigos de Padding:
#   0: valid, 1: same
#
# Capa 1: Conv1D (tipo=6), 32 filtros, kernel_size 2, activación 1 (relu), padding 1 (same)
6, 32, 2, 1, 1
# Capa 2: LSTM (tipo=3), 64 unidades, activación 1 (relu), return_sequences 1 (True)
3, 64, 1, 1, 0
# Capa 3: Dropout (tipo=4), rate 0.2
4, 0.2, 0, 0, 0
# Capa 4: Dense (tipo=1), 32 unidades, activación 1 (relu)
1, 32, 1, 0, 0
